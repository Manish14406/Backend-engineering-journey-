from sqlaclchemy.orm import Session 
def get_db():
    db = SessionLocal()    # opens a db session
    try:
        yield db           # serves till one endpoint
    finally:
        db.close()         # closes after each session 
**************************************************
Human understanding - (opens DB session,let the request use it,closes the session)
